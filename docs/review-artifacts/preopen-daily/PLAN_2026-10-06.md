`/etc/systemd/system/project-mai-tai-preopen.service`:

```ini
[Unit]
Description=Reviewed pinned daily preopen checks only
After=network-online.target
OnFailure=project-mai-tai-preopen-failure.service

[Service]
Type=oneshot
User=root
Group=root
WorkingDirectory=/home/trader/project-mai-tai
UnsetEnvironment=TZ
Environment=PYTHONDONTWRITEBYTECODE=1
ExecStart=/home/trader/preopen-daily/run.sh
TimeoutStartSec=300
Restart=no
UMask=0077
StandardOutput=journal
StandardError=journal
```

`/etc/systemd/system/project-mai-tai-preopen.timer`:

```ini
[Unit]
Description=Pinned preopen checks weekdays at0620 America/New_York

[Timer]
OnCalendar=Mon..Fri *-*-* 06:20:00 America/New_York
AccuracySec=1s
RandomizedDelaySec=0
Persistent=false
Unit=project-mai-tai-preopen.service

[Install]
WantedBy=timers.target
```

`/etc/systemd/system/project-mai-tai-preopen-failure.service`:
